option(OTAC_USE_P256 "Build the OpenSSL prime256v1 benchmark" OFF)
if(OTAC_USE_P256)
 enable_testing()
 function(add_p256_target target source)
  add_executable(${target} ${source})
  target_compile_definitions(${target} PRIVATE OTAC_USE_P256)
  target_include_directories(${target} PRIVATE "${GMP_INCLUDE_DIR}")
  target_link_libraries(${target} PRIVATE OpenSSL::Crypto "${GMP_LIBRARY}")
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
   target_compile_options(${target} PRIVATE -O3 -Wall -Wextra)
  endif()
 endfunction()

 add_p256_target(issuance_communication src/issuance_communication.cpp)
